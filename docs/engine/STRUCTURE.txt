│   ├── stages.py                   # preprocessing run_<stage>(ctx)
│   ├── analysis_stages.py          # tick_relevance, topic_model (MULTILINGUAL default),
│   │                               #   finalize_corpus, translate, layers 1-3
│   ├── screen.py                   # LLM screening + consensus + validation
│   ├── provenance.py               # run log + PRISMA + build_flows (dual diagrams)
│   ├── export.py                   # corpus_explore.xlsx + analysis_results.xlsx
│   ├── runner.py                   # orchestrates OPTIMIZED_ORDER + ANALYSIS_ORDER
│   └── cli.py                      # `sosci <stage>` entry point
│
├── configs/                        # ── PER-PROJECT configs ──
│   ├── project.example.yaml        # copy this for a new project
│   └── ticks_climate_ssa.yaml      # the tick study (multilingual + taxonomy wired)
│
├── projects/                       # ── PER-PROJECT assets (NOT engine code) ──

│   └── analysis/                   # standalone analysis + finalize/flows/multilingual/translate
│
├── docs/
